import socket

print("Testing TCP connections...")
targets = [
    ('127.0.0.1', 8080),
    ('127.0.0.1', 9222),
    ('127.0.0.1', 6379),
    ('127.0.0.1', 5432),
    ('127.0.0.1', 11211),
    ('127.0.0.1', 11300),
]

for host, port in targets:
    try:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(2)
        result = sock.connect_ex((host, port))
        sock.close()
        if result == 0:
            print(f"   ✅ {host}:{port}: CONNECTED")
        else:
            print(f"   ❌ {host}:{port}: FAILED")
    except Exception as e:
        print(f"   ❌ {host}:{port}: {e}")

print("\n✅ TCP connection test complete")
EOF
